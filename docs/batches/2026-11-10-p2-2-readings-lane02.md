```js
// node lane02-ruler.mjs（域 = 单一档；面 = 正文裸写）
const fs = require('fs');
const p = 'docs/core/design/DOC-DISCIPLINE.md';
const L = fs.readFileSync(p, 'utf8').split(/\r?\n/);
const FENCE = /^(\s*)(`{3,}|~{3,})/;
const BARE = ln => ln.replace(/(`+)[^`]*\1/g, '');
const CAND = /(?<![\w§.#/\-])(\d+)\s*(档|处|条|行|次|个|例|份|节|组|段|位|名|列|家|层|轮|对|款)/g;
const EXCL = ln => /[≤≥]/.test(ln) || /[<>]\s*\d|\d+\s*[<>]/.test(ln)
  || /软线|硬限|阈值|字符|上限/.test(ln) || /第\s*\d+\s*条/.test(ln) || /\d{4}-\d{2}/.test(ln);
let open = null, total = 0;
for (const ln of L) {
  const fm = FENCE.exec(ln);
  if (fm) { const ch = fm[2][0], len = fm[2].length;
    if (open === null) open = { ch, len }; else if (ch === open.ch && len >= open.len) open = null; continue; }
  if (open !== null) continue;
  if (EXCL(ln)) continue;
  total += [...BARE(ln).matchAll(CAND)].length;
}
console.log('候选', total);
```
